async function shareViaLiffTargetPicker() { const btn = document.getElementById('btnSharePicker'); const canvas = document.getElementById('cardCanvas'); const recipient = document.getElementById('targetRecipient').value.trim() || '好友'; const sender = document.getElementById('targetSender').value.trim() || '我'; const message = document.getElementById('targetMessage').value.trim(); btn.disabled = true; btn.innerHTML = ` 準備卡片中...`; try { // 嘗試上傳到 Supabase Storage if (!currentUploadedCardUrl) { try { const blob = await new Promise(resolve => canvas.toBlob(resolve, 'image/jpeg', 0.9)); const fileName = `generated_cards/${currentChurch}_${Date.now()}.jpg`; const { data, error } = await db.storage.from('church-media').upload(fileName, blob, { contentType: 'image/jpeg', cacheControl: '3600', upsert: true }); if (!error) { const { data: publicData } = db.storage.from('church-media').getPublicUrl(fileName); currentUploadedCardUrl = publicData.publicUrl; } } catch (uploadErr) { console.warn("Storage upload skipped or bucket missing:", uploadErr); } } // 判斷是否可用 LIFF shareTargetPicker if (window.liff && liff.isApiAvailable('shareTargetPicker')) { const sendMessages = []; if (currentUploadedCardUrl) { sendMessages.push({ type: "image", originalContentUrl: currentUploadedCardUrl, previewImageUrl: currentUploadedCardUrl }); } sendMessages.push({ type: "text", text: `💌 親愛的 ${recipient}：\n「${message}」\n\n— 來自 ${sender} 的誠摯祝禱 🌿` }); const res = await liff.shareTargetPicker(sendMessages); if (res) { alert('🎉 專屬祝賀卡已成功傳送給好友！'); } } else { // 降級跳轉分享 const shareText = encodeURIComponent(`💌 親愛的 ${recipient}：\n「${message}」\n\n— 來自 ${sender} 的誠摯祝禱 🌿`); window.location.href = `https://line.me/R/msg/text/?${shareText}`; } } catch (err) { console.error("Share error:", err); // 若使用者在 LINE 原生好友清單按了「取消」，不需跳警示視窗打擾使用者 if (err.code !== 'USER_CANCELLED') { alert('提示：請先在卡片上長按儲存圖片，即可在聊天室發給好友！'); } } finally { btn.disabled = false; btn.innerHTML = ` 選好友直接傳送`; } }
